
passengers = [
    ["Rahul", 25, 100],
    ["Priya", 12, 200],
    ["Ravi", 65, 300]
]

for p in passengers:
    name, age, distance = p

    if distance <= 100:
        fare = distance * 2
    elif distance <= 300:
        fare = distance * 1.5
    else:
        fare = distance * 1

    if age < 12:
        fare = fare * 0.5
    elif age >= 60:
        fare = fare * 0.7

    reservation = 50
    total = fare + reservation

    print("--------------------")
    print("Passenger:", name)
    print("Travel Distance:", distance, "km")
    print("Total Fare: Rs.", round(total, 2))

    cancel = input("Cancel ticket? (yes/no): ")

    if cancel == "yes":
        penalty = total * 0.20
        print("Refund: Rs.", round(total - penalty, 2))
    else:
        print("Ticket Confirmed")
